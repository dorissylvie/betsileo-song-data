"""
Construit le corpus ByT5 Betsileo <-> Malgache officiel.

Source unique :
nirina-doris-sylvie/scrapping-bible-protestante
  splits utilises : mg | betsileo   (fr et en sont ignores)

Changements par rapport a la version multilingue
------------------------------------------------
1. Plus de francais ni d'anglais. Une seule paire de langues, deux
   directions. Le modele ne disperse plus sa capacite sur des langues
   qui ne sont pas la cible.

2. Le decoupage train/validation/test se fait PAR LIVRE, pas au hasard.
   Un split aleatoire sur des versets met Matthieu 5:3 en train et
   Matthieu 5:4 en validation : les deux partagent vocabulaire, noms
   propres et tournures. Le score obtenu est alors une illusion. Avec un
   decoupage par livre, aucun verset du test n'a de voisin en train.

3. Deduplication, et comptage des paires ou le verset betsileo est
   identique au malgache officiel. Ces paires existent en nombre (les
   deux varietes sont proches) et gonflent artificiellement les scores :
   on les garde mais on les mesure, et le script d'evaluation les
   isole.

Usage :
  export HF_TOKEN=hf_...
  python 01_build_dataset_mg_bts.py
  python 01_build_dataset_mg_bts.py --no-push      # test local
"""

from __future__ import annotations

import argparse
import hashlib
import os
import re
from collections import Counter, defaultdict

DEFAULT_REPO = "nirina-doris-sylvie/scrapping-bible-protestante"
DEFAULT_OUTPUT_REPO = "nirina-doris-sylvie/byt5-betsileo-mg-ft"

MIN_CHARS = 10
MAX_CHARS = 800
SEED = 42

SRC_LANG = "Malagasy"
TGT_LANG = "Betsileo"

# Livres entierement reserves au test. Choisis pour couvrir des genres
# differents : recit, epitre, apocalyptique. Aucun verset de ces livres
# n'apparait en entrainement ni en validation.
TEST_BOOKS = {"MRK", "PHP", "JUD"}

# Livres reserves a la validation (suivi pendant l'entrainement).
VAL_BOOKS = {"TIT", "PHM", "2JN", "3JN"}


def rows_to_verse_index(rows) -> dict[tuple[str, str, str], str]:
    """Convertit un split HF (book, chapter, text[{verse: str}]) en index."""
    index: dict[tuple[str, str, str], str] = {}
    for row in rows:
        book = str(row.get("book") or "").strip().upper()
        chapter = row.get("chapter")
        if not book or chapter is None:
            continue
        try:
            chapter_i = int(chapter)
        except (TypeError, ValueError):
            continue
        if chapter_i <= 0:
            continue

        texts = row.get("text") or []
        for item in texts:
            if not isinstance(item, dict):
                continue
            for num_verset, texte in item.items():
                texte_propre = re.sub(r"\s+", " ", str(texte)).strip()
                if not texte_propre:
                    continue
                if not (MIN_CHARS <= len(texte_propre) <= MAX_CHARS):
                    continue
                index[(book, str(chapter_i), str(num_verset))] = texte_propre
    return index


def align_pairs(src_index: dict, tgt_index: dict) -> list[tuple[tuple, str, str]]:
    """Aligne sur la cle (livre, chapitre, verset). Garde la cle."""
    pairs = []
    for key, src_text in src_index.items():
        tgt_text = tgt_index.get(key)
        if tgt_text:
            pairs.append((key, src_text, tgt_text))
    return pairs


def deduplicate(pairs: list[tuple[tuple, str, str]]) -> tuple[list, int]:
    """Retire les doublons exacts (meme source ET meme cible)."""
    seen: set[str] = set()
    out = []
    dropped = 0
    for key, src, tgt in pairs:
        h = hashlib.sha1(f"{src}\u0000{tgt}".encode()).hexdigest()
        if h in seen:
            dropped += 1
            continue
        seen.add(h)
        out.append((key, src, tgt))
    return out, dropped


def assign_split(book: str) -> str:
    if book in TEST_BOOKS:
        return "test"
    if book in VAL_BOOKS:
        return "validation"
    return "train"


def pairs_to_examples(pairs: list[tuple[tuple, str, str]]) -> list[dict]:
    """Chaque paire alignee produit deux exemples : aller et retour."""
    examples = []
    for (book, chapter, verse), src, tgt in pairs:
        ref = f"{book} {chapter}:{verse}"
        identical = src.strip() == tgt.strip()
        examples.append({
            "source": f"translate {SRC_LANG} to {TGT_LANG}: {src}",
            "target": tgt,
            "direction": f"{SRC_LANG}->{TGT_LANG}",
            "ref": ref,
            "book": book,
            "identical": identical,
        })
        examples.append({
            "source": f"translate {TGT_LANG} to {SRC_LANG}: {tgt}",
            "target": src,
            "direction": f"{TGT_LANG}->{SRC_LANG}",
            "ref": ref,
            "book": book,
            "identical": identical,
        })
    return examples


def build(pairs: list[tuple[tuple, str, str]]) -> dict[str, list[dict]]:
    """Repartit les exemples en train / validation / test, par livre."""
    buckets: dict[str, list] = defaultdict(list)
    for item in pairs:
        buckets[assign_split(item[0][0])].append(item)
    return {name: pairs_to_examples(items) for name, items in buckets.items()}


def summarize(splits: dict[str, list[dict]]) -> None:
    print("\n  Repartition (decoupage par livre) :")
    for name in ("train", "validation", "test"):
        rows = splits.get(name, [])
        if not rows:
            print(f"    {name:<11} 0")
            continue
        books = sorted({r['book'] for r in rows})
        identical = sum(1 for r in rows if r["identical"])
        pct = 100 * identical / len(rows)
        print(f"    {name:<11} {len(rows):>6} exemples | "
              f"{len(books)} livres | "
              f"{pct:.1f}% de versets identiques mg/bts")
        if name != "train":
            print(f"                livres : {', '.join(books)}")

    all_books = {r["book"] for rows in splits.values() for r in rows}
    overlap = Counter()
    for name, rows in splits.items():
        for b in {r["book"] for r in rows}:
            overlap[b] += 1
    leaked = [b for b, c in overlap.items() if c > 1]
    if leaked:
        print(f"\n  ATTENTION : livres presents dans plusieurs splits : {leaked}")
    else:
        print(f"\n  Aucune fuite : les {len(all_books)} livres sont "
              f"dans un seul split.")

    main()

"""
Baseline de copie et caracterisation du corpus, AVANT tout entrainement.

Pourquoi ce script existe
-------------------------
Le corpus construit montre que les versets betsileo et malgache officiel
sont quasi identiques : 0,04 % de versets strictement egaux, mais
seulement quelques mots differents par verset. Exemple reel :

  mg  : nefa   hampamoahin'Ilay efa vonona  hitsara ny velona  sy ny maty
  bts : kanefa hampamoahin'Ilay efa vognona hitsara ny velogna sy ny maty

La tache n'est donc pas une traduction mais une transformation
systematique. Trois consequences :

1. Recopier l'entree obtient deja un score tres eleve. Il faut connaitre
   ce score AVANT d'entrainer, sinon on ne saura pas interpreter celui
   du modele.

2. Les metriques globales (chrF, BLEU) sont ecrasees par les mots
   inchanges. Un modele qui recopie tout obtient 85 de chrF en n'ayant
   rien appris. Ce script mesure donc aussi la performance sur les MOTS
   QUI CHANGENT, seule zone ou le modele a quelque chose a faire.

3. Les substitutions recurrentes (vonona -> vognona, velona -> velogna,
   nefa -> kanefa) constituent une grammaire differentielle empirique.
   Le script les extrait et les classe. C'est exploitable pour de
   l'augmentation par regles, et c'est un resultat presentable en soi.

Usage
-----
  python 00_baseline.py
  python 00_baseline.py --split test --top 60 --out baseline_results
"""

from __future__ import annotations

import argparse
import difflib
import json
import os
import re
import unicodedata
from collections import Counter

DEFAULT_DATASET = "nirina-doris-sylvie/byt5-betsileo-mg-ft"

WORD_RE = re.compile(r"[\w\u2019'-]+", re.UNICODE)


def tokenize(s: str) -> list[str]:
    return WORD_RE.findall(s)


def norm(s: str) -> str:
    """Normalise pour comparer : minuscules, apostrophes unifiees."""
    s = unicodedata.normalize("NFC", s)
    s = s.replace("\u2019", "'").replace("\u02bc", "'")
    return s.lower()


# ---------------------------------------------------------------------------
# Metriques
# ---------------------------------------------------------------------------
def corpus_metrics(hyps: list[str], refs: list[str]) -> dict:
    import sacrebleu

    pairs = [(h.strip(), r.strip())
             for h, r in zip(hyps, refs) if h.strip() and r.strip()]
    if not pairs:
        return {"n": 0, "chrf": 0.0, "bleu": 0.0, "exact": 0.0}
    hyp = [h for h, _ in pairs]
    ref = [[r for _, r in pairs]]
    return {
        "n": len(pairs),
        "chrf": round(sacrebleu.corpus_chrf(hyp, ref, word_order=2).score, 2),
        "bleu": round(sacrebleu.corpus_bleu(hyp, ref).score, 2),
        "exact": round(100 * sum(1 for h, r in pairs if h == r) / len(pairs), 2),
    }


def char_error_rate(hyps: list[str], refs: list[str]) -> float:
    """Taux d'erreur caractere, via distance d'edition (Levenshtein)."""
    total_edits, total_chars = 0, 0
    for h, r in zip(hyps, refs):
        h, r = h.strip(), r.strip()
        if not r:
            continue
        sm = difflib.SequenceMatcher(None, r, h, autojunk=False)
        matched = sum(b.size for b in sm.get_matching_blocks())
        total_edits += max(len(r), len(h)) - matched
        total_chars += len(r)
    return round(100 * total_edits / max(1, total_chars), 2)


# ---------------------------------------------------------------------------
# Analyse mot a mot
# ---------------------------------------------------------------------------
def word_alignment_stats(srcs: list[str], refs: list[str]
                         ) -> tuple[dict, Counter, Counter]:
    """
    Aligne les mots source et reference via difflib et compte :
      - la proportion de mots inchanges
      - les substitutions (mot mg -> mot bts)
      - les insertions et suppressions
    """
    kept = changed = inserted = deleted = 0
    subs: Counter = Counter()
    ops: Counter = Counter()

    for src, ref in zip(srcs, refs):
        a = [norm(t) for t in tokenize(src)]
        b = [norm(t) for t in tokenize(ref)]
        sm = difflib.SequenceMatcher(None, a, b, autojunk=False)
        for tag, i1, i2, j1, j2 in sm.get_opcodes():
            if tag == "equal":
                kept += i2 - i1
            elif tag == "replace":
                changed += max(i2 - i1, j2 - j1)
                ops["substitution"] += 1
                # Substitution mot pour mot quand les blocs ont la meme
                # taille : c'est le cas le plus informatif.
                if (i2 - i1) == (j2 - j1):
                    for k in range(i2 - i1):
                        subs[(a[i1 + k], b[j1 + k])] += 1
            elif tag == "insert":
                inserted += j2 - j1
                ops["insertion"] += 1
            elif tag == "delete":
                deleted += i2 - i1
                ops["deletion"] += 1

    total = kept + changed + inserted + deleted
    stats = {
        "mots_total": total,
        "mots_inchanges_pct": round(100 * kept / max(1, total), 2),
        "mots_substitues_pct": round(100 * changed / max(1, total), 2),
        "mots_inseres_pct": round(100 * inserted / max(1, total), 2),
        "mots_supprimes_pct": round(100 * deleted / max(1, total), 2),
    }
    return stats, subs, ops


def char_rules(subs: Counter, top: int = 25) -> Counter:
    """
    Deduit les transformations de caracteres recurrentes a partir des
    substitutions de mots. Donne la grammaire differentielle empirique.
    """
    rules: Counter = Counter()
    for (a, b), count in subs.items():
        sm = difflib.SequenceMatcher(None, a, b, autojunk=False)
        for tag, i1, i2, j1, j2 in sm.get_opcodes():
            if tag == "equal":
                continue
            src_frag = a[i1:i2] or "\u2205"
            tgt_frag = b[j1:j2] or "\u2205"
            # On ignore les fragments trop longs : ce sont des mots
            # entierement differents, pas des regles.
            if len(src_frag) <= 4 and len(tgt_frag) <= 4:
                rules[(src_frag, tgt_frag)] += count
    return rules

        print(f"  {a:<10} -> {b:<10} {n:>6}")
    results["char_rules"] = [
        {"mg": a, "bts": b, "count": n} for (a, b), n in rules.most_common(200)
    ]

    # ---- 5. exemples ----
    print("\n" + "-" * 66)
    print("5. EXEMPLES DE PAIRES  (mg -> betsileo)")
    print("-" * 66)
    for i in mg2bts[:5]:
        print(f"\n  mg  : {inputs[i][:96]}")
        print(f"  bts : {refs[i][:96]}")

    # ---- sorties ----

        for (a, b), n in rules.most_common():

----------------------------------------------------------------------
POURQUOI IL N'Y A PLUS DE SHARDING DES DONNEES
----------------------------------------------------------------------
L'ancien schema entrainait 5 epochs sur le 1er cinquieme des donnees,
puis 5 epochs sur le 2e cinquieme, etc. Ces sous-ensembles sont
disjoints : pendant les 5 epochs sur le shard 1, le modele n'a plus
aucun exemple du shard 0 et l'oublie en grande partie. C'est de l'oubli
catastrophique, et a la fin on obtient surtout un modele specialise sur
le dernier cinquieme.

La bonne facon de reprendre entre sessions est de decouper par EPOCH,
pas par sous-ensemble : chaque session fait 1 ou 2 passes sur la
TOTALITE des donnees, puis repart du checkpoint precedent.

  Session 1 : --epochs 2 --base-model google/byt5-base
  Session 2 : --epochs 2 --base-model doris-sylvie/byt5-betsileo-mg
  Session 3 : --epochs 2 --base-model doris-sylvie/byt5-betsileo-mg
  ...

Le modele voit tout le corpus a chaque session. Rien n'est oublie.
----------------------------------------------------------------------

# ---------------------------------------------------------------------------
# Push a chaque sauvegarde
# ---------------------------------------------------------------------------
class PushHubCallback(TrainerCallback):
    def __init__(self, repo_id: str, enabled: bool = True):
        self.repo_id = repo_id
        self.enabled = enabled

    def on_save(self, args, state, control, **kwargs):
        if not self.enabled or not state.is_world_process_zero:
            return
        model = kwargs.get("model")
        tok = kwargs.get("tokenizer") or kwargs.get("processing_class")
        if model is None:
            return
        try:
            model.push_to_hub(
                self.repo_id,
                commit_message=f"checkpoint step {state.global_step}",
                private=True,
            )
            if tok is not None:
                tok.push_to_hub(self.repo_id, private=True)
            print(f"  Push Hub @ step {state.global_step} -> {self.repo_id}")
        except Exception as e:
            print(f"  Push Hub echoue : {e}")


# ---------------------------------------------------------------------------
# Metrique pendant l'entrainement : chrF
# ---------------------------------------------------------------------------
class ChrfComputer:
    """
    chrF plutot que BLEU comme metrique de suivi.

    BLEU compte des n-grammes de MOTS. Sur une langue agglutinante a
    morphologie riche et sur un modele qui travaille au niveau de
    l'octet, un mot presque juste vaut zero en BLEU. chrF compte des
    n-grammes de CARACTERES et capture ces quasi-reussites, ce qui en
    fait une metrique bien plus informative ici. On garde BLEU dans
    03_evaluate.py pour la comparabilite avec la litterature.
    """

    def __init__(self, tokenizer, pad_id: int):
        self.tokenizer = tokenizer
        self.pad_id = pad_id

    def __call__(self, eval_pred):
        import sacrebleu

        predictions, labels = eval_pred
        if isinstance(predictions, tuple):
            predictions = predictions[0]
        if getattr(predictions, "ndim", 0) == 3:
            predictions = np.argmax(predictions, axis=-1)

        predictions = np.where(predictions != -100, predictions, self.pad_id)
        labels = np.where(labels != -100, labels, self.pad_id)

        try:
            preds = self.tokenizer.batch_decode(
                predictions, skip_special_tokens=True)
            refs = self.tokenizer.batch_decode(
                labels, skip_special_tokens=True)
        except Exception as e:
            print(f"  decode error: {e}")
            return {"chrf": 0.0}

        valid = [(p.strip(), r.strip())
                 for p, r in zip(preds, refs) if p.strip() and r.strip()]
        if not valid:
            return {"chrf": 0.0}

        print("\n  -- Exemples generation --")
        for p, r in valid[:3]:
            print(f"    Predit    : {p[:76]}")
            print(f"    Reference : {r[:76]}")

        # [refs] et non [[r] for r in refs] : sacrebleu attend une
        # liste de FLUX de references, pas une liste de listes a une
        # entree. La confusion fait tomber le score a zero.
        chrf = sacrebleu.corpus_chrf(
            [p for p, _ in valid], [[r for _, r in valid]], word_order=2
        )
        return {"chrf": round(chrf.score, 2)}


# ---------------------------------------------------------------------------
# Sanity check
# ---------------------------------------------------------------------------
def sanity_check_forward(model, tokenizer, device):
    model.train()
    src = "translate Malagasy to Betsileo: Ny filazana ny razan'i Jesosy Kristy"
    tgt = "Hoatsa izao ro tetiaran'ny razamben'i Jesoa Kristy"
    inputs = tokenizer(src, return_tensors="pt", truncation=True,
                       max_length=512).to(device)
    labels = tokenizer(text_target=tgt, return_tensors="pt", truncation=True,
                       max_length=256).to(device)
    with torch.no_grad():
        out = model(**inputs, labels=labels["input_ids"])
    loss = float(out.loss.item())
    print(f"  Forward loss = {loss:.4f}")
    if loss == 0.0 or np.isnan(loss):
        raise ValueError(f"Forward pass anormal (loss={loss})")
    print("  Forward pass OK")


# ---------------------------------------------------------------------------
# Entrainement
# ---------------------------------------------------------------------------
def run_training(
    *,
    dataset: str = DEFAULT_DATASET,
    base_model: str = "google/byt5-base",
    output_repo: str = DEFAULT_OUTPUT_REPO,
    output_dir: str = "./byt5-betsileo-mg",
    epochs: int = 2,
    batch_size: int = 4,
    grad_accum: int = 4,
    learning_rate: float = 3e-4,
    max_source_length: int = 512,
    max_target_length: int = 256,
    num_workers: int = 2,
    push: bool = True,
    eval_max_samples: int = 256,
    chrf_during_train: bool = True,
    early_stopping_patience: int = 3,
):
    install_deps()
    setup_hf_auth()

    device = "cuda" if torch.cuda.is_available() else "cpu"
    n_gpu = torch.cuda.device_count() if device == "cuda" else 0

    print("=" * 62)
    print("TRAIN ByT5 - Betsileo <-> Malgache officiel")
    print("=" * 62)
    print(f"Device      : {device} | GPUs : {n_gpu}")
    if device == "cuda":
        print(f"GPU 0       : {torch.cuda.get_device_name(0)}")
    print(f"Base model  : {base_model}")
    print(f"Epochs      : {epochs}  (sur la TOTALITE du corpus)")
    print(f"LR          : {learning_rate}")

    # ---- modele ----
    print("\n[1/5] Chargement modele...")
    config = AutoConfig.from_pretrained(base_model)
    if "google/byt5" in base_model:
        config.tie_word_embeddings = False
    tokenizer = AutoTokenizer.from_pretrained(base_model)
    model = AutoModelForSeq2SeqLM.from_pretrained(base_model, config=config)
    model = model.to(device)

    if hasattr(model, "gradient_checkpointing_enable"):
        model.gradient_checkpointing_enable()
        model.config.use_cache = False
        print("  gradient_checkpointing ON")

    print(f"  Parametres : {sum(p.numel() for p in model.parameters()):,}")
    print("\n  -- Sanity check --")
    sanity_check_forward(model, tokenizer, device)
    pad_id = tokenizer.pad_token_id

    # ---- donnees ----
    print(f"\n[2/5] Dataset : {dataset}")
    raw = load_dataset(dataset)
    train_raw = raw["train"]
    eval_raw = raw.get("validation") or raw.get("test")
    if eval_raw is None:
        split = train_raw.train_test_split(test_size=0.1, seed=42)
        train_raw, eval_raw = split["train"], split["test"]

    print(f"  Train={len(train_raw)} | Eval={len(eval_raw)}")
    if len(eval_raw) > eval_max_samples:
        eval_raw = eval_raw.shuffle(seed=42).select(range(eval_max_samples))
        print(f"  Eval echantillonnee a {len(eval_raw)} (vitesse)")
    print(f"  Colonnes : {train_raw.column_names}")

    def preprocess(batch):
        model_inputs = tokenizer(
            batch["source"], max_length=max_source_length,
            truncation=True, padding=False)
        labels = tokenizer(
            text_target=batch["target"], max_length=max_target_length,
            truncation=True, padding=False)
        model_inputs["labels"] = labels["input_ids"]
        return model_inputs

    print("\n[3/5] Tokenisation (padding dynamique)...")
    train_tok = train_raw.map(
        preprocess, batched=True, batch_size=128,
        remove_columns=list(train_raw.column_names), desc="tokenize train")
    eval_tok = eval_raw.map(
        preprocess, batched=True, batch_size=128,
        remove_columns=list(eval_raw.column_names), desc="tokenize eval")

    if not train_tok[0]["labels"]:
        raise ValueError("Labels vides apres tokenisation")

    def add_length(batch):
        batch["length"] = [len(x) for x in batch["input_ids"]]
        return batch

    train_tok = train_tok.map(add_length, batched=True)
    eval_tok = eval_tok.map(add_length, batched=True)

    collator = DataCollatorForSeq2Seq(
        tokenizer=tokenizer, model=model, padding=True,
        label_pad_token_id=-100)

    eff_batch = batch_size * grad_accum * max(1, n_gpu)
    steps_per_epoch = max(1, len(train_tok) // eff_batch)

    print(f"\n[4/5] Config")
    print(f"  effective batch = {eff_batch}")
    print(f"  steps/epoch     = {steps_per_epoch}")
    print(f"  total steps     = {steps_per_epoch * epochs}")

    training_args = Seq2SeqTrainingArguments(
        output_dir=output_dir,
        eval_strategy="epoch",
        save_strategy="epoch",
        load_best_model_at_end=True,
        per_device_train_batch_size=batch_size,
        per_device_eval_batch_size=batch_size,
        gradient_accumulation_steps=grad_accum,
        # Adafactor sans learning rate explicite utilise un schedule
        # interne qui convient mal au fine-tuning court. On fixe le LR.
        learning_rate=learning_rate,
        lr_scheduler_type="linear",
        warmup_ratio=0.03,
        weight_decay=0.0,
        num_train_epochs=epochs,
        fp16=False,
        bf16=False,
        max_grad_norm=1.0,
        generation_max_length=max_target_length,
        logging_steps=max(10, steps_per_epoch // 10),
        logging_first_step=True,
        optim="adafactor",
        save_total_limit=2,
        report_to="none",
        dataloader_num_workers=num_workers,
        dataloader_pin_memory=(device == "cuda"),
        remove_unused_columns=True,
        group_by_length=True,
        length_column_name="length",
    )

    if chrf_during_train:
        training_args.predict_with_generate = True
        training_args.generation_num_beams = 1
        training_args.metric_for_best_model = "chrf"
        training_args.greater_is_better = True
        compute_metrics = ChrfComputer(tokenizer, pad_id)
        print("  Suivi : chrF a chaque epoch")
    else:
        training_args.predict_with_generate = False
        training_args.metric_for_best_model = "loss"
        training_args.greater_is_better = False
        compute_metrics = None
        print("  Suivi : loss seulement (plus rapide)")

    callbacks = [EarlyStoppingCallback(
        early_stopping_patience=early_stopping_patience)]
    if push:
        callbacks.append(PushHubCallback(output_repo, enabled=True))

    trainer = Seq2SeqTrainer(
        model=model,
        args=training_args,
        train_dataset=train_tok,
        eval_dataset=eval_tok,
        processing_class=tokenizer,
        data_collator=collator,
        compute_metrics=compute_metrics,
        callbacks=callbacks,
    )

    print("\n[5/5] Entrainement...")
    print("=" * 62)
    trainer.train()

    print("\nSauvegarde...")
    trainer.save_model(output_dir)
    tokenizer.save_pretrained(output_dir)

    if push:
        print(f"\nPush final -> {output_repo}")
        model.push_to_hub(output_repo, private=True,
                          commit_message=f"+{epochs} epochs")
        tokenizer.push_to_hub(output_repo, private=True)
        print(f"  https://huggingface.co/{output_repo}")
        print("  Session suivante : relancer avec")
        print(f"    --base-model {output_repo}")

    print("\nTermine. Evaluation : python 03_evaluate.py")


def parse_args():
    p = argparse.ArgumentParser()
    p.add_argument("--dataset", default=DEFAULT_DATASET)
    p.add_argument("--base-model", default="google/byt5-base")
    p.add_argument("--output-repo", default=DEFAULT_OUTPUT_REPO)
    p.add_argument("--output-dir", default="./byt5-betsileo-mg")
    p.add_argument("--epochs", type=int, default=2)
    p.add_argument("--batch-size", type=int, default=4)
    p.add_argument("--grad-accum", type=int, default=4)
    p.add_argument("--learning-rate", type=float, default=3e-4)
    p.add_argument("--max-source-length", type=int, default=512)
    p.add_argument("--max-target-length", type=int, default=256)
    p.add_argument("--num-workers", type=int, default=2)
    p.add_argument("--no-push", action="store_true")
    p.add_argument("--no-chrf", action="store_true",
                   help="desactive chrF pendant le train (plus rapide)")
    p.add_argument("--eval-max-samples", type=int, default=256)
    args, _ = p.parse_known_args()

    for env, attr, cast in (("BASE_MODEL", "base_model", str),
                            ("EPOCHS", "epochs", int),
                            ("LR", "learning_rate", float)):
        if os.environ.get(env):
            setattr(args, attr, cast(os.environ[env]))
    return args


def main():
    a = parse_args()
    run_training(
        dataset=a.dataset,
        base_model=a.base_model,
        output_repo=a.output_repo,
        output_dir=a.output_dir,
        epochs=a.epochs,
        batch_size=a.batch_size,
        grad_accum=a.grad_accum,
        learning_rate=a.learning_rate,
        max_source_length=a.max_source_length,
        max_target_length=a.max_target_length,
        num_workers=a.num_workers,
        push=not a.no_push,
        eval_max_samples=a.eval_max_samples,
