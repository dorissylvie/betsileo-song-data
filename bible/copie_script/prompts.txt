# --- 3. Définition de la Normalisation ---

def normalize_betsimisaraka(text):
    if not isinstance(text, str):
        return ""

    text = text.lower().strip()

    # 1. Standardisation des caractères
    text = text.replace('ô', 'o')


    text = text.replace('ñ', 'gn')

    # 2. Standardisation des variations lexicales courantes

    replacements = {
        "zagny": "zegny",
        "izegny": "zegny",
        "zany": "zegny",
        "izany": "zegny",
        "ôlogno": "olo",
        "ologno": "olo",
        "olona": "olo",
    }

    words = text.split()
    new_words = [replacements.get(word, word) for word in words]
    return " ".join(new_words)

def compute_metrics(eval_preds):
    preds, labels = eval_preds
    if isinstance(preds, tuple):
        preds = preds[0]

    # Décoder les prédictions
    decoded_preds = tokenizer.batch_decode(preds, skip_special_tokens=True)

    # Remplacer -100 par pad_token_id pour décoder les labels
    labels = np.where(labels != -100, labels, tokenizer.pad_token_id)
    decoded_labels = tokenizer.batch_decode(labels, skip_special_tokens=True)

    # Post-traitement simple
    decoded_preds = [pred.strip() for pred in decoded_preds]
    decoded_labels = [[label.strip()] for label in decoded_labels]

    result = metric.compute(predictions=decoded_preds, references=decoded_labels)
    return {"bleu": result["score"]}

# A. Normalisation de l'entrée Betsimisaraka (pour BMM -> MLG)
# On aide le modèle à comprendre les variantes en entrée
def clean_input_betsimisaraka(text):
    if not isinstance(text, str): return ""
    text = text.lower().strip()
    text = text.replace('ô', 'o').replace('ñ', 'gn')
    # On unifie les entrées pour réduire le vocabulaire
    replacements = {"zagny": "zegny", "zany": "zegny", "ôlogno": "ôlo", "olona": "ôlo", "tragno": "trano"}
    words = text.split()
    return " ".join([replacements.get(w, w) for w in words])

# B. Normalisation de la CIBLE Betsimisaraka (pour MLG -> BMM)
# C'est CRUCIAL : On force le modèle à apprendre UNE SEULE orthographe correcte
def normalize_target_betsimisaraka(text):
    if not isinstance(text, str): return ""
    text = text.lower().strip()
    text = text.replace('ô', 'o').replace('ñ', 'gn')

    # Règles strictes de génération
    replacements = {
        # Le modèle doit apprendre à dire "ôlo" (ou olo selon votre choix) et pas "olona"
        "olona": "ôlo", "ologno": "ôlo", "ôlogno": "ôlo",
        # Le modèle doit apprendre "zegny"
        "izany": "zegny", "zagny": "zegny", "zany": "zegny", "izegny": "zegny",
        # Autres
        "trano": "tragno", # Si vous voulez trano
    }
    words = text.split()
    return " ".join([replacements.get(w, w) for w in words])

    preds, labels = eval_preds

    decoded_labels = [[l] for l in decoded_labels]

def translate_universal(text, direction, model, tokenizer):
    """
    direction: 'to_merina' ou 'to_betsimisaraka'
    """
    # 1. Préparer le prompt selon la direction
    if direction == 'to_merina':
        # On normalise l'entrée Betsimisaraka
        clean_text = clean_input_betsimisaraka(text)
        prompt = f"translate Betsimisaraka to Malagasy: {clean_text}"
    else:
        # L'entrée Merina est juste nettoyée
        clean_text = text.lower().strip()
        prompt = f"translate Malagasy to Betsimisaraka: {clean_text}"

    # 2. Inférence
    inputs = tokenizer(prompt, return_tensors="pt", max_length=128, truncation=True)
    if torch.cuda.is_available():
        inputs = {k: v.to('cuda') for k, v in inputs.items()}
        model.to('cuda')

    outputs = model.generate(**inputs, max_length=128, num_beams=5, early_stopping=True)
    return tokenizer.decode(outputs[0], skip_special_tokens=True)

def generate_predictions(model, tokenizer, dataset):

    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    model.to(device)
    model.eval()

    references = []

    print(f"Génération des prédictions sur {len(dataset)} exemples...")

        # Préparation du prompt pour la tâche BMM -> MLG (DOIT être le même que pour l'entraînement)

            num_beams=5,

def translate(text, model, tokenizer, direction):
    prompt = f"translate {direction}: {text}"
    inputs = tokenizer(prompt, return_tensors="pt", truncation=True, max_length=256)
    if torch.cuda.is_available():
        inputs = {k: v.to("cuda") for k, v in inputs.items()}

    outputs = model.generate(
        **inputs,
        max_length=256,
        num_beams=4,
        early_stopping=True
    )
    return tokenizer.decode(outputs[0], skip_special_tokens=True)

bleu = evaluate.load("sacrebleu")
chrf = evaluate.load("chrf")

def compute_metrics(preds, refs):
    bleu_score = bleu.compute(predictions=preds, references=[[r] for r in refs])["score"]
    chrf_score = chrf.compute(predictions=preds, references=refs)["score"]
    return bleu_score, chrf_score


results = []

def evaluate_model(use_cases, model, tokenizer, direction, label):
    for uc, texts in use_cases.items():
        preds = []
        refs = []

        for text in texts:
            pred = translate(text, model, tokenizer, direction)
            preds.append(pred)
            refs.append(text)  # proxy reference (qualitative focus)

        bleu_s, chrf_s = compute_metrics(preds, refs)

        results.append({
            "Modèle": label,
            "Use case": uc,
            "BLEU": round(bleu_s, 2),
            "chrF": round(chrf_s, 2)
        })
